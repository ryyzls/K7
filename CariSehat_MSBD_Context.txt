CREATE OR REPLACE FUNCTION fn_hari_ini_wib() RETURNS date
LANGUAGE sql STABLE AS $$ SELECT (now() AT TIME ZONE 'Asia/Jakarta')::date $$;

-- Kapasitas kamar per RS & jenis. Bila belum diatur admin, default 5 kamar.
CREATE OR REPLACE FUNCTION fn_kapasitas_kamar(p_rs text, p_jenis text) RETURNS integer
LANGUAGE sql STABLE AS $$
  SELECT COALESCE((SELECT kapasitas FROM kuota_kamar WHERE rs_id = p_rs AND jenis_kamar = p_jenis), 5)
$$;

-- Okupansi puncak (jumlah kamar terpakai terbanyak pada satu hari) di periode tertentu.
CREATE OR REPLACE FUNCTION fn_terisi_kamar(p_rs text, p_jenis text, p_masuk date, p_keluar date, p_exclude integer DEFAULT NULL)
RETURNS integer LANGUAGE sql STABLE AS $$
  SELECT COALESCE(MAX(t.c), 0)::integer FROM (
    SELECT g.d::date AS hari, COUNT(b.id) AS c
    FROM generate_series(p_masuk::timestamp,
                         LEAST(COALESCE(p_keluar, p_masuk), p_masuk + 365)::timestamp,
                         interval '1 day') AS g(d)
    LEFT JOIN booking_kamar b
      ON b.kode_rs = p_rs AND b.jenis_kamar = p_jenis AND b.status <> 'dibatalkan'
     AND (p_exclude IS NULL OR b.id <> p_exclude)
     AND g.d::date BETWEEN b.tanggal_masuk AND COALESCE(b.tanggal_keluar, b.tanggal_masuk)
    GROUP BY g.d
  ) t
$$;

CREATE OR REPLACE FUNCTION fn_sisa_kamar(p_rs text, p_jenis text, p_masuk date, p_keluar date, p_exclude integer DEFAULT NULL)
RETURNS integer LANGUAGE sql STABLE AS $$
  SELECT GREATEST(fn_kapasitas_kamar(p_rs, p_jenis) - fn_terisi_kamar(p_rs, p_jenis, p_masuk, p_keluar, p_exclude), 0)
$$;

-- Stored procedure booking ATOMIK: kunci advisory per (RS, jenis kamar) agar dua
-- request bersamaan tidak bisa sama-sama mengambil kamar terakhir (race condition).
CREATE OR REPLACE FUNCTION sp_buat_booking(
  p_kode text, p_rs text, p_pasien text, p_telp text,
  p_masuk date, p_keluar date, p_jenis text, p_catatan text
) RETURNS integer LANGUAGE plpgsql AS $$
DECLARE v_id integer;
BEGIN
  PERFORM pg_advisory_xact_lock(hashtextextended('booking:' || p_rs || ':' || p_jenis, 0));
  IF fn_sisa_kamar(p_rs, p_jenis, p_masuk, p_keluar, NULL) <= 0 THEN
    RAISE EXCEPTION 'Kamar % penuh pada periode yang dipilih', p_jenis USING ERRCODE = 'CS409';
  END IF;
  INSERT INTO booking_kamar (kode_booking, kode_rs, nama_pasien, no_telp, tanggal_masuk, tanggal_keluar, jenis_kamar, catatan)
  VALUES (p_kode, p_rs, p_pasien, p_telp, p_masuk, p_keluar, p_jenis, p_catatan)
  RETURNING id INTO v_id;
  RETURN v_id;
END $$;

-- Sisa kuota satu sesi konsultasi pada tanggal tertentu.
CREATE OR REPLACE FUNCTION fn_sisa_kuota_konsultasi(p_jadwal integer, p_tanggal date) RETURNS integer
LANGUAGE sql STABLE AS $$
  SELECT GREATEST(j.kuota_per_sesi - (
           SELECT COUNT(*) FROM reservasi_konsultasi r
           WHERE r.jadwal_id = j.id AND r.tanggal = p_tanggal AND r.status <> 'dibatalkan'
         )::integer, 0)
  FROM jadwal_konsultasi j WHERE j.id = p_jadwal
$$;

-- Reservasi konsultasi ATOMIK dengan validasi hari praktik, sesi, dan kuota.
CREATE OR REPLACE FUNCTION sp_buat_reservasi_konsultasi(
  p_kode text, p_jadwal integer, p_pasien text, p_telp text,
  p_tanggal date, p_keluhan text, p_idem text
) RETURNS integer LANGUAGE plpgsql AS $$
DECLARE
  j jadwal_konsultasi%ROWTYPE;
  v_id integer;
  v_sekarang timestamp := (now() AT TIME ZONE 'Asia/Jakarta');
BEGIN
  PERFORM pg_advisory_xact_lock(hashtextextended('konsultasi:' || p_jadwal || ':' || p_tanggal, 0));
  SELECT * INTO j FROM jadwal_konsultasi WHERE id = p_jadwal AND aktif;
  IF NOT FOUND THEN
    RAISE EXCEPTION 'Jadwal konsultasi tidak ditemukan atau tidak aktif' USING ERRCODE = 'CS404';
  END IF;
  IF EXTRACT(ISODOW FROM p_tanggal)::integer <> j.hari THEN
    RAISE EXCEPTION 'Tanggal tidak sesuai hari praktik dokter' USING ERRCODE = 'CS422';
  END IF;
  IF p_tanggal < v_sekarang::date OR (p_tanggal = v_sekarang::date AND v_sekarang::time >= j.jam_selesai) THEN
    RAISE EXCEPTION 'Sesi konsultasi sudah berakhir atau tanggal sudah lewat' USING ERRCODE = 'CS422';
  END IF;
  IF fn_sisa_kuota_konsultasi(p_jadwal, p_tanggal) <= 0 THEN
    RAISE EXCEPTION 'Kuota sesi konsultasi pada tanggal tersebut sudah penuh' USING ERRCODE = 'CS409';
  END IF;
  INSERT INTO reservasi_konsultasi (kode_reservasi, jadwal_id, rs_id, nama_pasien, no_telp, tanggal, keluhan, idempotency_key)
  VALUES (p_kode, j.id, j.rs_id, p_pasien, p_telp, p_tanggal, p_keluhan, p_idem)
  RETURNING id INTO v_id;
  RETURN v_id;
END $$;

CREATE OR REPLACE FUNCTION fn_set_updated_at() RETURNS trigger
LANGUAGE plpgsql AS $$ BEGIN NEW.updated_at := NOW(); RETURN NEW; END $$;

CREATE OR REPLACE FUNCTION fn_audit() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE
  v_old jsonb; v_new jsonb; v_actor text;
BEGIN
  v_actor := COALESCE(NULLIF(current_setting('app.aktor', true), ''), 'system');
  IF TG_OP IN ('UPDATE', 'DELETE') THEN v_old := to_jsonb(OLD) - 'password_hash'; END IF;
  IF TG_OP IN ('INSERT', 'UPDATE') THEN v_new := to_jsonb(NEW) - 'password_hash'; END IF;
  INSERT INTO audit_log (aktor, aksi, tabel, record_id, data_lama, data_baru)
  VALUES (v_actor, TG_OP, TG_TABLE_NAME, COALESCE(v_new, v_old) ->> TG_ARGV[0], v_old, v_new);
  IF TG_OP = 'DELETE' THEN RETURN OLD; END IF;
  RETURN NEW;
END $$;

CREATE OR REPLACE FUNCTION fn_booking_validasi() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  IF NEW.tanggal_keluar IS NOT NULL AND NEW.tanggal_keluar < NEW.tanggal_masuk THEN
    RAISE EXCEPTION 'tanggal_keluar tidak boleh sebelum tanggal_masuk' USING ERRCODE = 'CS422';
  END IF;
  IF TG_OP = 'UPDATE' AND OLD.status = 'dibatalkan' AND NEW.status <> 'dibatalkan' THEN
    RAISE EXCEPTION 'Booking yang sudah dibatalkan tidak dapat diaktifkan kembali' USING ERRCODE = 'CS422';
  END IF;
  RETURN NEW;
END $$;

CREATE OR REPLACE FUNCTION fn_jadwal_validasi() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  IF NEW.aktif AND EXISTS (
    SELECT 1 FROM jadwal_konsultasi o
    WHERE o.rs_id = NEW.rs_id AND o.nama_dokter = NEW.nama_dokter AND o.hari = NEW.hari AND o.aktif
      AND o.id IS DISTINCT FROM NEW.id
      AND o.jam_mulai < NEW.jam_selesai AND NEW.jam_mulai < o.jam_selesai
  ) THEN
    RAISE EXCEPTION 'Jadwal dokter bentrok dengan jadwal lain di hari yang sama' USING ERRCODE = 'CS409';
  END IF;
  RETURN NEW;
END $$;

DROP TRIGGER IF EXISTS trg_booking_validasi ON booking_kamar;
CREATE TRIGGER trg_booking_validasi BEFORE INSERT OR UPDATE ON booking_kamar
  FOR EACH ROW EXECUTE FUNCTION fn_booking_validasi();

DROP TRIGGER IF EXISTS trg_jadwal_validasi ON jadwal_konsultasi;

CREATE OR REPLACE FUNCTION sp_buat_reservasi_poliklinik(
  p_kode    text,
  p_jadwal  integer,
  p_user_id integer,
  p_pasien  text,
  p_telp    text,
  p_tanggal date,
  p_keluhan text,
  p_idem    text
) RETURNS TABLE(reservasi_id integer, nomor_antrean integer, kode_tiket text)
LANGUAGE plpgsql AS $$
DECLARE
  j           jadwal_konsultasi%ROWTYPE;
  p           poliklinik%ROWTYPE;
  v_id        integer;
  v_antrean   integer;
  v_tiket     text;
  v_terisi_h  integer;
  v_sekarang  timestamp := (now() AT TIME ZONE 'Asia/Jakarta');
BEGIN
  -- Kunci transaksi per (jadwal, tanggal) agar nomor antrean & kuota 100% atomik
  PERFORM pg_advisory_xact_lock(hashtextextended('poli_antrean:' || p_jadwal || ':' || p_tanggal, 0));

  SELECT * INTO j FROM jadwal_konsultasi WHERE id = p_jadwal AND aktif;
  IF NOT FOUND THEN
    RAISE EXCEPTION 'Jadwal dokter tidak ditemukan atau sedang nonaktif' USING ERRCODE = 'CS404';
  END IF;

  IF j.poliklinik_id IS NOT NULL THEN
    SELECT * INTO p FROM poliklinik WHERE id = j.poliklinik_id;
    IF FOUND AND NOT p.aktif THEN
      RAISE EXCEPTION 'Unit poliklinik sedang ditutup sementara' USING ERRCODE = 'CS422';
    END IF;
  END IF;

  IF EXTRACT(ISODOW FROM p_tanggal)::integer <> j.hari THEN
    RAISE EXCEPTION 'Tanggal kunjungan tidak sesuai hari praktik dokter' USING ERRCODE = 'CS422';
  END IF;

  IF p_tanggal < v_sekarang::date OR (p_tanggal = v_sekarang::date AND v_sekarang::time >= j.jam_selesai) THEN
    RAISE EXCEPTION 'Sesi praktik sudah berakhir atau tanggal sudah lewat' USING ERRCODE = 'CS422';
  END IF;

  -- Cek kuota sesi dokter
  IF fn_sisa_kuota_konsultasi(p_jadwal, p_tanggal) <= 0 THEN
    RAISE EXCEPTION 'Kuota sesi dokter pada tanggal tersebut sudah penuh' USING ERRCODE = 'CS409';
  END IF;

  -- Cek kuota harian unit poliklinik (jika terhubung ke master poliklinik)
  IF p.id IS NOT NULL THEN
    SELECT COUNT(*)::integer INTO v_terisi_h
    FROM reservasi_konsultasi r
    JOIN jadwal_konsultasi jk ON jk.id = r.jadwal_id
    WHERE jk.poliklinik_id = p.id
      AND r.tanggal = p_tanggal
      AND r.status <> 'dibatalkan';

    IF v_terisi_h >= p.kuota_harian THEN
      RAISE EXCEPTION 'Kuota harian % sudah penuh', p.nama_poli USING ERRCODE = 'CS409';
    END IF;
  END IF;

  -- Hitung nomor antrean berikutnya secara atomik
  SELECT COALESCE(MAX(r.nomor_antrean), 0) + 1 INTO v_antrean
  FROM reservasi_konsultasi r
  WHERE r.jadwal_id = p_jadwal AND r.tanggal = p_tanggal;

  v_tiket := 'TKT-' || to_char(p_tanggal, 'YYYYMMDD') || '-' || nextval('seq_tiket_poliklinik');

  INSERT INTO reservasi_konsultasi (
    kode_reservasi, jadwal_id, rs_id, user_id, nama_pasien, no_telp,
    tanggal, keluhan, idempotency_key, nomor_antrean, kode_tiket
  )
  VALUES (
    p_kode, j.id, j.rs_id, p_user_id, p_pasien, p_telp,
    p_tanggal, p_keluhan, p_idem, v_antrean, v_tiket
  )
  RETURNING id INTO v_id;

  RETURN QUERY SELECT v_id, v_antrean, v_tiket;
END $$;

-- View Peringkat RS (menggunakan CTE, agregasi AVG/COUNT, HAVING, dan Window Functions)
CREATE OR REPLACE VIEW v_peringkat_rs AS
WITH agregat_ulasan AS (
  SELECT
    rs_id,
    ROUND(AVG(rating)::numeric, 2) AS rata_rating,
    COUNT(*)::integer              AS total_ulasan,
    COUNT(*) FILTER (WHERE rating = 5)::integer AS jumlah_bintang_5
  FROM ulasan_rs
  WHERE status_moderasi = 'disetujui'
  GROUP BY rs_id
  HAVING COUNT(*) >= 1
),
agregat_favorit AS (
  SELECT rs_id, COUNT(*)::integer AS total_favorit
  FROM favorit_rs
  GROUP BY rs_id
)
SELECT
  u.rs_id,
  u.rata_rating,
  u.total_ulasan,
  u.jumlah_bintang_5,
  COALESCE(f.total_favorit, 0) AS total_favorit,
  RANK() OVER (ORDER BY u.rata_rating DESC, u.total_ulasan DESC)       AS peringkat_rating,
  DENSE_RANK() OVER (ORDER BY u.total_ulasan DESC, u.rata_rating DESC) AS peringkat_terpopuler
FROM agregat_ulasan u
LEFT JOIN agregat_favorit f ON f.rs_id = u.rs_id;

"""Runner migrasi SQL sederhana (tanpa Alembic) untuk PostgreSQL.

- Membaca database/migrations/NNN_nama.up.sql, menjalankan yang belum tercatat
  di tabel schema_migrations, masing-masing dalam SATU transaksi.
- Hanya nomor >= 3 yang otomatis dijalankan. Migrasi 001/002 adalah legacy manual
  dan efeknya sudah diserap (idempotent) oleh 003.
- Memakai advisory lock agar dua proses (mis. uvicorn --reload) tidak berebut.
"""
from __future__ import annotations

import logging
import re
from pathlib import Path

from app.config import settings

log = logging.getLogger("carisehat.migrate")

AUTO_FROM = 3
_LOCK_ID = 727001
_NAME_RE = re.compile(r"^(\d{3})_(.+)\.up\.sql$")


def find_migrations_dir() -> Path | None:
    candidates = [
        Path(settings.migrations_dir),
        Path(__file__).resolve().parents[3] / "database" / "migrations",
        Path(__file__).resolve().parents[2] / "database" / "migrations",
    ]
    for c in candidates:
        if c.is_dir():
            return c
    return None


def run_migrations(engine) -> list[str]:
    """Jalankan migrasi tertunda. Mengembalikan daftar versi yang baru diterapkan."""
    directory = find_migrations_dir()
    if directory is None:
        log.warning("Folder migrations tidak ditemukan; fitur admin butuh database/migrations (lihat docker-compose volume).")
        return []

    files = []
    for f in sorted(directory.glob("*.up.sql")):
        m = _NAME_RE.match(f.name)
        if m and int(m.group(1)) >= AUTO_FROM:
            files.append((f.name[: -len(".up.sql")], f))

    applied_now: list[str] = []
    raw = engine.raw_connection()
    try:
        cur = raw.cursor()
        cur.execute("SELECT pg_advisory_lock(%s)", (_LOCK_ID,))
        cur.execute(
            "CREATE TABLE IF NOT EXISTS schema_migrations ("
            "version TEXT PRIMARY KEY, applied_at TIMESTAMPTZ NOT NULL DEFAULT NOW())"
        )
        raw.commit()
        cur.execute("SELECT version FROM schema_migrations")
        done = {r[0] for r in cur.fetchall()}
        raw.commit()

        for version, path in files:
            if version in done:
                continue
            sql = path.read_text(encoding="utf-8")
            try:
                cur.execute(sql)  # tanpa parameter: karakter % di PL/pgSQL aman
                cur.execute("INSERT INTO schema_migrations (version) VALUES (%s)", (version,))
                raw.commit()
                applied_now.append(version)
                log.info("Migrasi diterapkan: %s", version)
            except Exception:
                raw.rollback()
                log.exception("Migrasi %s GAGAL dan di-rollback; migrasi berikutnya dihentikan.", version)
                break
    finally:
        try:
            raw.rollback()
            cur = raw.cursor()
            cur.execute("SELECT pg_advisory_unlock(%s)", (_LOCK_ID,))
            raw.commit()
        except Exception:  # pragma: no cover - koneksi sudah rusak
            pass
        raw.close()
    return applied_now

    if not hospitals:

                if spec and key not in seen_spec:
                    seen_spec.add(key)
                    unique_specs.append(spec)
                if len(unique_specs) >= 4:
                    break

            if not unique_specs:
                unique_specs = ["Poli Umum"]

@contextmanager
def tx(db: Session):
    """Transaksi: commit bila sukses, rollback + terjemahkan error DB ke HTTP bila gagal."""
    try:
        yield
        db.commit()
    except HTTPException:
        db.rollback()
        raise
    except IntegrityError as exc:
        db.rollback()
        c = getattr(getattr(exc.orig, "diag", None), "constraint_name", "") or ""
        raise HTTPException(409, _CONSTRAINT_MSG.get(c, "Data melanggar aturan integritas database")) from None
    except DBAPIError as exc:
        db.rollback()
        orig = getattr(exc, "orig", None)
        code = getattr(orig, "pgcode", "") or ""
        msg = getattr(getattr(orig, "diag", None), "message_primary", "") or ""
        if code in _PGCODE_STATUS:
            raise HTTPException(_PGCODE_STATUS[code], msg or "Ditolak oleh aturan database") from None
        raise HTTPException(503, "Database tidak tersedia; coba lagi nanti") from None
    except SQLAlchemyError:
        db.rollback()
        raise HTTPException(503, "Database tidak tersedia; coba lagi nanti") from None

def _with_names(rows, key: str = "kode_rs") -> list[dict]:
    cache: dict[str, str] = {}
    out = []
    for r in rows:
        d = dict(r)
        rid = d.get(key)
        if rid not in cache:
            cache[rid] = hospital_name(rid) or rid
        d["nama_rs"] = cache[rid]
        out.append(d)
    return out

@router.post("/login", summary="Login admin, mengembalikan token Bearer")
def login(body: LoginIn, db: Session = Depends(get_db)):
    global _DUMMY_HASH
    if _DUMMY_HASH is None:
        _DUMMY_HASH = hash_password("dummy-password")
    with tx(db):
        row = db.execute(
            text("SELECT id, username, nama_lengkap, role_kode, password_hash, aktif, failed_attempts, "
                 "(locked_until IS NOT NULL AND locked_until > now()) AS locked "
                 "FROM admin_users WHERE lower(username) = lower(:u)"), {"u": body.username}).mappings().first()
        if row and row["locked"]:
            raise HTTPException(429, f"Akun dikunci sementara {settings.admin_lock_minutes} menit karena terlalu banyak percobaan gagal")
        ok = verify_password(body.password, row["password_hash"] if row else _DUMMY_HASH)
        if not (row and row["aktif"] and ok):
            if row:
                db.execute(text("SELECT set_config('app.aktor', :a, true)"), {"a": row["username"]})
                db.execute(
                    text("UPDATE admin_users SET failed_attempts = failed_attempts + 1, "
                         "locked_until = CASE WHEN failed_attempts + 1 >= :mx THEN now() + make_interval(mins => :mn) ELSE locked_until END "
                         "WHERE id = :id"),
                    {"id": row["id"], "mx": settings.admin_max_failed_logins, "mn": settings.admin_lock_minutes})
                db.commit()
            raise HTTPException(401, "Username atau password salah")
        token = secrets.token_urlsafe(32)
        db.execute(text("SELECT set_config('app.aktor', :a, true)"), {"a": row["username"]})
        db.execute(text("DELETE FROM admin_sessions WHERE expires_at < now()"))
        db.execute(
            text("INSERT INTO admin_sessions (token_hash, user_id, expires_at) "
                 "VALUES (:h, :u, now() + make_interval(hours => :hrs))"),
            {"h": hash_token(token), "u": row["id"], "hrs": settings.admin_session_hours})
        db.execute(text("UPDATE admin_users SET failed_attempts = 0, locked_until = NULL, last_login = now() WHERE id = :id"),
                   {"id": row["id"]})
        perms = db.execute(text("SELECT permission FROM role_permissions WHERE role_kode = :r"),
                           {"r": row["role_kode"]}).scalars().all()
    return {"token": token, "expires_in": settings.admin_session_hours * 3600,
            "user": {"id": row["id"], "username": row["username"], "nama_lengkap": row["nama_lengkap"],
                     "role_kode": row["role_kode"], "permissions": sorted(perms)}}

        SELECT
            d.tgl::text AS tanggal,

        FROM days d
        ORDER BY d.tgl ASC
    """)).mappings().all()

        "top_rs": top,

@router.post("/bookings", status_code=201, summary="Buat booking (lewat sp_buat_booking: cek kuota atomik)")
def create_booking(body: BookingIn, user: dict = Depends(require_perm("booking.write")), db: Session = Depends(get_db)):
    _need_rs(body.kode_rs)
    with tx(db):
        set_actor(db, user)
        new_id = None
        for _ in range(5):
            try:
                with db.begin_nested():
                    new_id = db.execute(
                        text("SELECT sp_buat_booking(:kode,:rs,:pasien,:telp,:masuk,:keluar,:jenis,:catatan)"),
                        dict(kode=_gen("BK"), rs=body.kode_rs, pasien=body.nama_pasien, telp=body.no_telp,
                             masuk=body.tanggal_masuk, keluar=body.tanggal_keluar, jenis=body.jenis_kamar,
                             catatan=body.catatan)).scalar_one()
                break
            except IntegrityError as exc:
                c = getattr(getattr(exc.orig, "diag", None), "constraint_name", "") or ""
                if "kode_booking" not in c:
                    raise
        if new_id is None:
            raise HTTPException(409, "Gagal membuat kode booking unik; coba lagi")
        if body.status != "menunggu":
            db.execute(text("UPDATE booking_kamar SET status = :s WHERE id = :id"), {"s": body.status, "id": new_id})
    return get_booking(new_id, user, db)


@router.put("/bookings/{bid}", summary="Ubah booking; kuota dicek ulang bila kamar/tanggal berubah")
def update_booking(bid: int, body: BookingPatch, user: dict = Depends(require_perm("booking.write")),
                   db: Session = Depends(get_db)):
    changes = body.model_dump(exclude_unset=True)
    if "kode_rs" in changes:
        _need_rs(changes["kode_rs"])
    with tx(db):
        set_actor(db, user)
        cur = db.execute(text("SELECT * FROM booking_kamar WHERE id = :id FOR UPDATE"), {"id": bid}).mappings().first()
        if not cur:
            raise HTTPException(404, "Booking tidak ditemukan")
        new = {**dict(cur), **changes}
        if new["tanggal_keluar"] and new["tanggal_keluar"] < new["tanggal_masuk"]:
            raise HTTPException(422, "tanggal_keluar harus >= tanggal_masuk")
        affects = any(k in changes for k in ("kode_rs", "jenis_kamar", "tanggal_masuk", "tanggal_keluar"))
        if new["status"] != "dibatalkan" and affects:
            db.execute(text("SELECT pg_advisory_xact_lock(hashtextextended(:k, 0))"),
                       {"k": f"booking:{new['kode_rs']}:{new['jenis_kamar']}"})
            sisa = db.execute(text("SELECT fn_sisa_kamar(:rs,:j,:m,:k,:ex)"),
                              {"rs": new["kode_rs"], "j": new["jenis_kamar"], "m": new["tanggal_masuk"],
                               "k": new["tanggal_keluar"], "ex": bid}).scalar_one()
            if sisa <= 0:
                raise HTTPException(409, f"Kamar {new['jenis_kamar']} penuh pada periode yang dipilih")
        db.execute(
            text("UPDATE booking_kamar SET kode_rs=:kode_rs, nama_pasien=:nama_pasien, no_telp=:no_telp, "
                 "tanggal_masuk=:tanggal_masuk, tanggal_keluar=:tanggal_keluar, jenis_kamar=:jenis_kamar, "
                 "catatan=:catatan, status=:status WHERE id=:id"),
            {k: new[k] for k in ("kode_rs", "nama_pasien", "no_telp", "tanggal_masuk", "tanggal_keluar",
                                 "jenis_kamar", "catatan", "status")} | {"id": bid})
    return get_booking(bid, user, db)

    catatan: str = Field(default="", max_length=200)


def _ensure_igd_status_table(db: Session) -> None:
    db.execute(text("""
        CREATE TABLE IF NOT EXISTS status_igd_rs (
            rs_id VARCHAR(20) PRIMARY KEY,
            status_igd VARCHAR(16) NOT NULL DEFAULT 'normal',
            catatan TEXT NOT NULL DEFAULT '',
            updated_by VARCHAR(60) NOT NULL DEFAULT 'system',
            updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
        )
    """))


@router.patch("/kuota/step", summary="Ubah cepat kapasitas kamar (+1 / -1) secara atomik")
def step_kuota(body: KuotaStepIn, user: dict = Depends(require_perm("kuota.write")), db: Session = Depends(get_db)):
    _need_rs(body.rs_id)
    with tx(db):
        set_actor(db, user)
        cur = db.execute(
            text("SELECT kapasitas FROM kuota_kamar WHERE rs_id = :rs AND jenis_kamar = :j"),
            {"rs": body.rs_id, "j": body.jenis_kamar},
        ).scalar_one_or_none()
        base_cap = int(cur) if cur is not None else 5
        new_cap = max(0, min(1000, base_cap + body.delta))
        db.execute(
            text("""
                INSERT INTO kuota_kamar (rs_id, jenis_kamar, kapasitas)
                VALUES (:rs, :j, :k)
                ON CONFLICT (rs_id, jenis_kamar) DO UPDATE SET kapasitas = EXCLUDED.kapasitas
            """),
            {"rs": body.rs_id, "j": body.jenis_kamar, "k": new_cap},
        )
    return {"message": f"Kapasitas {body.jenis_kamar} diperbarui ke {new_cap}", "kapasitas": new_cap}

def _enforce_rs_scope(user: dict, requested_rs: str = "") -> str:
    scoped = (user.get("scoped_rs_id") or "").strip()
    if scoped:
        if requested_rs and requested_rs != scoped:
            raise HTTPException(403, f"Akun ini hanya memiliki akses operasional untuk RS '{scoped}'.")
        return scoped
    return requested_rs

@router.put("/users/{uid}")
def update_user(uid: int, body: UserPatch, user: dict = Depends(require_perm("user.manage")),
                db: Session = Depends(get_db)):
    ch = body.model_dump(exclude_unset=True)
    with tx(db):
        set_actor(db, user)
        cur = db.execute(text("SELECT id, role_kode, aktif FROM admin_users WHERE id = :id FOR UPDATE"), {"id": uid}).mappings().first()
        if not cur:
            raise HTTPException(404, "Pengguna tidak ditemukan")
        if "role_kode" in ch:
            _check_role(db, ch["role_kode"])
        new_role = ch.get("role_kode", cur["role_kode"])
        new_aktif = ch.get("aktif", cur["aktif"])
        if uid == user["id"] and (not new_aktif or new_role != "superadmin"):
            raise HTTPException(400, "Anda tidak dapat menonaktifkan atau menurunkan role akun sendiri")
        if cur["role_kode"] == "superadmin" and cur["aktif"] and (new_role != "superadmin" or not new_aktif) \
                and _other_superadmins(db, uid) == 0:
            raise HTTPException(400, "Minimal harus ada satu superadmin aktif")
        if "nama_lengkap" in ch:
            db.execute(text("UPDATE admin_users SET nama_lengkap = :v WHERE id = :id"), {"v": ch["nama_lengkap"], "id": uid})
        db.execute(text("UPDATE admin_users SET role_kode = :r, aktif = :a WHERE id = :id"),
                   {"r": new_role, "a": new_aktif, "id": uid})
        if ch.get("password"):
            db.execute(text("UPDATE admin_users SET password_hash = :p, failed_attempts = 0, locked_until = NULL WHERE id = :id"),
                       {"p": hash_password(ch["password"]), "id": uid})
        if uid != user["id"] and (ch.get("password") or "aktif" in ch or "role_kode" in ch):
            db.execute(text("DELETE FROM admin_sessions WHERE user_id = :id"), {"id": uid})  # paksa login ulang
    return {"message": "Pengguna diperbarui", "id": uid}

def _as_dict(val):
    if val is None:
        return None
    if isinstance(val, str):
        return json.loads(val)
    return dict(val)


@router.post("/audit/{audit_id}/rollback", summary="Pulihkan data berdasarkan catatan audit_log (1-Click Rollback)")
def rollback_audit_entry(
    audit_id: int,
    user: dict = Depends(require_perm("audit.read")),
    db: Session = Depends(get_db),
):
    row = db.execute(
        text("SELECT id, aksi, tabel, record_id, data_lama, data_baru FROM audit_log WHERE id = :id"),
        {"id": audit_id},
    ).mappings().first()
    if not row:
        raise HTTPException(status_code=404, detail="Catatan audit tidak ditemukan.")

    tabel = row["tabel"]
    aksi = (row["aksi"] or "").upper()
    req_perm = _ROLLBACK_TABLE_PERM.get(tabel)
    if not req_perm:
        raise HTTPException(status_code=400, detail=f"Modul '{tabel}' tidak mendukung rollback otomatis.")

    perms = set(user.get("permissions") or [])
    if req_perm not in perms:
        raise HTTPException(status_code=403, detail=f"Membutuhkan izin '{req_perm}' untuk memulihkan modul {tabel}.")

    data_lama = _as_dict(row["data_lama"])
    data_baru = _as_dict(row["data_baru"])

    col_rows = db.execute(
        text("""
            SELECT column_name
            FROM information_schema.columns
            WHERE table_schema = 'public' AND table_name = :tbl
            ORDER BY ordinal_position
        """),
        {"tbl": tabel},
    ).fetchall()
    valid_cols = [c[0] for c in col_rows]
    if not valid_cols:
        raise HTTPException(status_code=400, detail=f"Skema tabel '{tabel}' tidak ditemukan.")

    try:
        set_actor(db, user)
        if aksi == "UPDATE":
            if not data_lama:
                raise HTTPException(status_code=400, detail="Snapshot data_lama kosong, tidak bisa dipulihkan.")
            if tabel == "kuota_kamar":
                pk_where = "rs_id = :pk_rs AND jenis_kamar = :pk_jk"
                params = {"pk_rs": data_lama["rs_id"], "pk_jk": data_lama["jenis_kamar"]}
                skip = {"rs_id", "jenis_kamar", "created_at"}
            else:
                pk_where = "id = :pk_id"
                params = {"pk_id": data_lama["id"]}
                skip = {"id", "created_at"}

            set_parts = []
            for col in valid_cols:
                if col in skip or col not in data_lama:
                    continue
                if col == "updated_at":
                    set_parts.append("updated_at = NOW()")
                else:
                    set_parts.append(f"{col} = :v_{col}")
                    params[f"v_{col}"] = data_lama[col]

            res = db.execute(text(f"UPDATE {tabel} SET {', '.join(set_parts)} WHERE {pk_where}"), params)
            if res.rowcount == 0:
                raise HTTPException(status_code=404, detail="Baris target sudah tidak ada di database.")

        elif aksi == "DELETE":
            if not data_lama:
                raise HTTPException(status_code=400, detail="Snapshot data_lama kosong, baris terhapus tidak bisa dipulihkan.")
            cols = [c for c in valid_cols if c in data_lama]
            placeholders = [f":v_{c}" for c in cols]
            params = {f"v_{c}": data_lama[c] for c in cols}
            db.execute(
                text(f"INSERT INTO {tabel} ({', '.join(cols)}) VALUES ({', '.join(placeholders)})"),
                params,
            )

        elif aksi == "INSERT":
            if not data_baru:
                raise HTTPException(status_code=400, detail="Snapshot data_baru kosong, data baru tidak bisa dibatalkan.")
            if tabel == "kuota_kamar":
                res = db.execute(
                    text("DELETE FROM kuota_kamar WHERE rs_id = :rs AND jenis_kamar = :jk"),
                    {"rs": data_baru["rs_id"], "jk": data_baru["jenis_kamar"]},
                )
            else:
                res = db.execute(text(f"DELETE FROM {tabel} WHERE id = :id"), {"id": data_baru["id"]})
            if res.rowcount == 0:
                raise HTTPException(status_code=404, detail="Baris target sudah dihapus sebelumnya.")
        else:
            raise HTTPException(status_code=400, detail=f"Jenis tindakan '{aksi}' tidak didukung.")

        db.commit()
    except HTTPException:
        db.rollback()
        raise
    except Exception as exc:
        db.rollback()
        raise HTTPException(status_code=400, detail=f"Gagal memulihkan data: {exc}")

    return {"ok": True, "message": f"Data {tabel} ({row['record_id']}) berhasil dipulihkan dari audit #{audit_id}."}

def _hash_password(password: str) -> str:
    salt = os.urandom(16)
    dk = hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), salt, 120_000)
    return f"pbkdf2_sha256$120000${salt.hex()}${dk.hex()}"


def _verify_password(password: str, stored_hash: str) -> bool:
    try:
        _, iter_str, salt_hex, dk_hex = stored_hash.split("$", 3)
        dk = hashlib.pbkdf2_hmac(
            "sha256", password.encode("utf-8"), bytes.fromhex(salt_hex), int(iter_str)
        )
        return hmac.compare_digest(dk.hex(), dk_hex)
    except Exception:
        return False


def _hash_token(raw_token: str) -> str:
    return hashlib.sha256(raw_token.encode("utf-8")).hexdigest()


def get_optional_user(
    authorization: Optional[str] = Header(default=None),
    db: Session = Depends(get_db),
) -> Optional[User]:
    if not authorization or not authorization.startswith("Bearer "):
        return None
    raw_token = authorization.removeprefix("Bearer ").strip()
    if not raw_token:
        return None
    token_hash = _hash_token(raw_token)
    now = datetime.now(timezone.utc)
    stmt = (
        select(User)
        .join(UserSession, UserSession.user_id == User.id)
        .where(
            UserSession.token_hash == token_hash,
            UserSession.expires_at > now,
            User.aktif.is_(True),
        )
    )
    return db.scalar(stmt)


def require_user(user: Optional[User] = Depends(get_optional_user)) -> User:
    if user is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Silakan login terlebih dahulu untuk mengakses fitur ini.",
        )
    return user

    ident = payload.username_or_email.strip()
    stmt = select(User).where(
        (User.username == ident) | (User.email == ident.lower()),
        User.aktif.is_(True),
    )
    user = db.scalar(stmt)
    if user is None or not _verify_password(payload.password, user.password_hash):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Username/email atau kata sandi salah.",
        )

    raw_token = secrets.token_urlsafe(32)
    session_row = UserSession(
        token_hash=_hash_token(raw_token),
        user_id=user.id,
        expires_at=datetime.now(timezone.utc) + timedelta(hours=SESSION_TTL_HOURS),
    )
    db.add(session_row)
    db.commit()

    return {
        "access_token": raw_token,
        "token_type": "bearer",
        "user": {
            "id": user.id,
            "username": user.username,
            "email": user.email,
            "nama_lengkap": user.nama_lengkap,
            "no_telp": user.no_telp,
        },
    }

    code = getattr(orig, "pgcode", "") or ""
    diag = getattr(orig, "diag", None)
    msg = getattr(diag, "message_primary", "") or ""
    return code, msg

async def verified_booking(data: BookingCreate) -> BookingCreate:
    data._request_hash = hashlib.sha256(json.dumps(data.model_dump(mode="json"), sort_keys=True, separators=(",", ":"), ensure_ascii=False).encode()).hexdigest()
    hospital = await sp.get_hospital(data.kode_rs)

@router.post("/booking", status_code=201, response_model=BookingResponse,
             summary="Buat booking kamar rawat inap (transaksi ACID)")
def buat_booking(data: BookingCreate = Depends(verified_booking), db: Session = Depends(get_db),
                 idempotency_key: Annotated[str | None, Header(min_length=16, max_length=128, pattern=r"^[A-Za-z0-9_-]+$")] = None,
                 current_user: User | None = Depends(get_optional_user)):
    """Simpan booking ke PostgreSQL. Ledger idempotency + stored function sp_buat_booking
    berjalan dalam SATU transaksi; kegagalan apa pun di-rollback."""
    for _ in range(5):
        kode = _generate_kode()
        try:
            params = None
            if idempotency_key:
                key_hash = hashlib.sha256(idempotency_key.encode()).hexdigest()
                payload_hash = data._request_hash or hashlib.sha256(json.dumps(data.model_dump(mode="json"), sort_keys=True, separators=(",", ":"), ensure_ascii=False).encode()).hexdigest()
                params = dict(scope=_SCOPE, key_hash=key_hash, payload_hash=payload_hash)
                # INSERT pesaing menunggu pemilik kunci commit/rollback (tidak ada reservasi setengah jadi).
                owner = db.execute(text("INSERT INTO booking_idempotency (scope,key_hash,payload_hash) VALUES (:scope,:key_hash,:payload_hash) ON CONFLICT (scope,key_hash) DO NOTHING RETURNING key_hash"), params).scalar()
                if owner is None:
                    row = db.execute(text("SELECT payload_hash,booking_id FROM booking_idempotency WHERE scope=:scope AND key_hash=:key_hash"), params).mappings().one()
                    if row["payload_hash"] != payload_hash:
                        db.rollback()
                        raise HTTPException(409, "Idempotency-Key telah digunakan dengan payload berbeda")
                    previous = db.get(BookingKamar, row["booking_id"]) if row["booking_id"] else None
                    if previous is None:
                        db.rollback()
                        raise HTTPException(503, "Hasil booking belum tersedia")
                    if settings.idempotency_replay_as_409:
                        db.rollback()
                        raise HTTPException(409, f"Idempotency-Key sudah dipakai; booking {previous.kode_booking} sudah tercatat")
                    response = _response(previous, data.nama_rs)
                    db.rollback()
                    return response
            # Hanya operasi baru yang dicek: replay sah boleh datang di hari berikutnya.
            if data.tanggal_masuk < datetime.date.today():
                db.rollback()
                raise HTTPException(422, "tanggal_masuk tidak boleh di masa lalu")
            new_id = db.execute(
                text("SELECT sp_buat_booking(:kode,:rs,:pasien,:telp,:masuk,:keluar,:jenis,:catatan)"),
                dict(kode=kode, rs=data.kode_rs, pasien=data.nama_pasien, telp=data.no_telp,
                     masuk=data.tanggal_masuk, keluar=data.tanggal_keluar, jenis=data.jenis_kamar,
                     catatan=data.catatan),
            ).scalar_one()
            if current_user is not None:
                # Tautkan booking ke akun pasien yang login (transaksi yang sama) agar
                # muncul di Dashboard Pasien. Tamu tetap user_id = NULL.
                db.execute(text("UPDATE booking_kamar SET user_id=:uid WHERE id=:id"),
                           dict(uid=current_user.id, id=new_id))
            if params:
                db.execute(text("UPDATE booking_idempotency SET booking_id=:booking_id WHERE scope=:scope AND key_hash=:key_hash"), dict(params, booking_id=new_id))
            booking = db.get(BookingKamar, new_id)
            # Materialisasi sebelum commit agar kegagalan baca tidak memicu insert ulang.
            response = _response(booking, data.nama_rs)
            db.commit()
            return response
        except HTTPException:
            raise
        except IntegrityError as exc:
            db.rollback()
            constraint = getattr(getattr(exc.orig, "diag", None), "constraint_name", "") or ""
            if "kode_booking" in constraint:
                continue
            if "uq_booking_pasien_aktif" in constraint:
                raise HTTPException(409, "Pasien yang sama sudah memiliki booking aktif untuk kamar dan tanggal tersebut") from None
            raise HTTPException(409, "Data booking tidak memenuhi aturan database") from None
        except DBAPIError as exc:
            db.rollback()
            code, msg = _pg_error(exc)
            if code in _PGCODE_STATUS:
                raise HTTPException(_PGCODE_STATUS[code], msg or "Booking ditolak oleh aturan database") from None
            raise HTTPException(503, "Database booking tidak tersedia; coba lagi nanti") from None
        except SQLAlchemyError:
            db.rollback()
            raise HTTPException(503, "Database booking tidak tersedia; coba lagi nanti") from None
    raise HTTPException(409, "Gagal membuat kode booking unik; coba lagi")


def _operator_ok(x_booking_key: str) -> bool:
    return bool(settings.booking_access_key) and secrets.compare_digest(
        x_booking_key.encode(), settings.booking_access_key.encode()
    )


def require_booking_access(x_booking_key: str = Header(default="")):
    if not _operator_ok(x_booking_key):
        raise HTTPException(403, "Akses ini hanya untuk operator simulasi yang berwenang")

@router.get("/booking/{kode_booking}", response_model=BookingResponse,
            summary="Cek status booking (kode + nomor telepon pemesan, atau kunci operator)")
def cek_booking(kode_booking: str, db: Session = Depends(get_db),
                x_booking_key: str = Header(default=""),
                no_telp: Annotated[str | None, Query(max_length=20)] = None):
    b = db.query(BookingKamar).filter(BookingKamar.kode_booking == kode_booking.upper()).first()
    proof = bool(no_telp) and b is not None and secrets.compare_digest(
        _digits(no_telp).encode(), _digits(b.no_telp).encode()
    ) and bool(_digits(no_telp))
    if not (_operator_ok(x_booking_key) or proof):
        # Respons sama untuk "tidak ada" dan "bukan pemilik" agar kode tidak bisa ditebak.
        raise HTTPException(403, "Kode booking dan nomor telepon pemesan tidak cocok")
    if b is None:
        raise HTTPException(404, "Kode booking tidak ditemukan")
    return _response(b)

        b = db.query(BookingKamar).filter(BookingKamar.kode_booking == kode_booking.upper()).first()
        if not b:
            raise HTTPException(404, "Kode booking tidak ditemukan")
        if b.status == "dibatalkan":
            raise HTTPException(400, "Booking sudah dibatalkan sebelumnya")
        db.execute(text("SELECT set_config('app.aktor', 'operator-api', true)"))
        b.status = "dibatalkan"
        db.commit()
    except HTTPException:
        db.rollback()
        raise
    except SQLAlchemyError:
        db.rollback()
        raise HTTPException(503, "Database booking tidak tersedia; coba lagi nanti") from None
    return {"message": "Booking berhasil dibatalkan", "kode_booking": kode_booking}

def _kode() -> str:
    suffix = "".join(secrets.choice(string.ascii_uppercase + string.digits) for _ in range(8))
    return f"KS-{datetime.date.today():%Y%m%d}-{suffix}"


def _digits(v: str) -> str:
    return "".join(c for c in v if c.isdigit())


@router.get("/{kode_rs}", summary="Jadwal praktik dokter di RS; beri ?tanggal= untuk sisa kuota pada hari itu")
def jadwal_rs(kode_rs: str, tanggal: datetime.date | None = None, db: Session = Depends(get_db)):
    if not hospital_name(kode_rs):
        raise HTTPException(404, "Rumah sakit tidak ditemukan dalam dataset semantik")
    rows = db.execute(
        text(
            "SELECT j.id, j.nama_dokter, j.spesialisasi, j.hari, "
            "to_char(j.jam_mulai,'HH24:MI') AS jam_mulai, to_char(j.jam_selesai,'HH24:MI') AS jam_selesai, "
            "j.kuota_per_sesi, "
            "CASE WHEN CAST(:t AS date) IS NULL THEN NULL ELSE fn_sisa_kuota_konsultasi(j.id, CAST(:t AS date)) END AS sisa "
            "FROM jadwal_konsultasi j "
            "WHERE j.rs_id = :rs AND j.aktif "
            "AND (CAST(:t AS date) IS NULL OR j.hari = EXTRACT(ISODOW FROM CAST(:t AS date))::int) "
            "ORDER BY j.hari, j.jam_mulai"
        ),
        {"rs": kode_rs, "t": tanggal},
    ).mappings().all()
    return {"kode_rs": kode_rs, "tanggal": tanggal, "items": [dict(r) for r in rows]}

"""Autentikasi & otorisasi sistem admin (tanpa dependensi tambahan).

- Password: PBKDF2-HMAC-SHA256 + salt acak (hashlib, stdlib).
- Sesi: token acak 256-bit; yang disimpan di DB hanya SHA-256-nya (admin_sessions).
- RBAC: role -> permission di tabel role_permissions; dicek via require_perm().
- Semua query SQL diparameterisasi (text() + bind), tidak ada f-string ke SQL.
"""
from __future__ import annotations

import hashlib
import hmac
import logging
import secrets

from fastapi import Depends, Header, HTTPException
from sqlalchemy import text
from sqlalchemy.orm import Session

from app.config import settings
from app.database.connection import engine, get_db

log = logging.getLogger("carisehat.auth")
_ITERATIONS = 210_000


def hash_password(password: str) -> str:
    salt = secrets.token_bytes(16)
    digest = hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), salt, _ITERATIONS)
    return f"pbkdf2_sha256${_ITERATIONS}${salt.hex()}${digest.hex()}"


def verify_password(password: str, stored: str) -> bool:
    try:
        algo, iterations, salt_hex, hash_hex = stored.split("$")
        if algo != "pbkdf2_sha256":
            return False
        digest = hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), bytes.fromhex(salt_hex), int(iterations))
        return hmac.compare_digest(digest.hex(), hash_hex)
    except (ValueError, TypeError):
        return False


def hash_token(token: str) -> str:
    return hashlib.sha256(token.encode("utf-8")).hexdigest()


def ensure_default_admin() -> None:
    """Buat akun admin awal bila tabel admin_users masih kosong."""
    with engine.begin() as conn:
        total = conn.execute(text("SELECT COUNT(*) FROM admin_users")).scalar_one()
        if total:
            return
        conn.execute(
            text(
                "INSERT INTO admin_users (username, password_hash, nama_lengkap, role_kode) "
                "VALUES (:u, :p, 'Administrator Utama', 'superadmin')"
            ),
            {"u": settings.admin_username, "p": hash_password(settings.admin_password)},
        )
    log.warning("Akun admin awal dibuat: username=%s (segera ganti password di menu Pengguna).", settings.admin_username)


def ensure_demo_operator() -> None:
    """Buat akun demo ber-role 'operator' (idempotent: dilewati bila username sudah ada).

    Operator hanya mendapat permission dari role_permissions (booking, kuota kamar,
    jadwal, reservasi/poliklinik); tidak bisa mengelola pengguna, hapus data, atau audit log.
    """
    if not settings.demo_operator_enabled:
        return
    with engine.begin() as conn:
        created = conn.execute(
            text(
                "INSERT INTO admin_users (username, password_hash, nama_lengkap, role_kode) "
                "SELECT :u, :p, 'Operator Demo', 'operator' "
                "WHERE EXISTS (SELECT 1 FROM roles WHERE kode = 'operator') "
                "ON CONFLICT (username) DO NOTHING RETURNING id"
            ),
            {"u": settings.operator_username, "p": hash_password(settings.operator_password)},
        ).scalar()
    if created:
        log.warning("Akun demo operator dibuat: username=%s", settings.operator_username)


def current_admin(authorization: str = Header(default=""), db: Session = Depends(get_db)) -> dict:
    scheme, _, token = authorization.partition(" ")
    if scheme.lower() != "bearer" or not token.strip():
        raise HTTPException(401, "Login admin diperlukan", headers={"WWW-Authenticate": "Bearer"})
    try:
        db.execute(text("ALTER TABLE admin_users ADD COLUMN IF NOT EXISTS scoped_rs_id VARCHAR(20)"))
        db.commit()
    except Exception:
        db.rollback()
    row = db.execute(
        text(
            "SELECT u.id, u.username, u.nama_lengkap, u.role_kode, u.scoped_rs_id "
            "FROM admin_sessions s JOIN admin_users u ON u.id = s.user_id "
            "WHERE s.token_hash = :h AND s.expires_at > now() AND u.aktif"
        ),
        {"h": hash_token(token.strip())},
    ).mappings().first()
    if not row:
        raise HTTPException(401, "Sesi tidak valid atau sudah berakhir", headers={"WWW-Authenticate": "Bearer"})
    perms = db.execute(
        text("SELECT permission FROM role_permissions WHERE role_kode = :r"), {"r": row["role_kode"]}
    ).scalars().all()
    return {**dict(row), "permissions": set(perms)}


def require_perm(permission: str):
    def dependency(user: dict = Depends(current_admin)) -> dict:
        if permission not in user["permissions"]:
            raise HTTPException(403, f"Hak akses '{permission}' diperlukan untuk aksi ini")
        return user

    return dependency


def set_actor(db: Session, user: dict) -> None:
    """Catat pelaku perubahan agar trigger audit_log mengisinya (berlaku per transaksi)."""
    db.execute(text("SELECT set_config('app.aktor', :a, true)"), {"a": user["username"]})

## 4. Perbaikan idempotency
Kontrak header opsional Idempotency-Key: 16–128 karakter ASCII alfanumerik, underscore atau hyphen. Anjurkan UUID acak baru per operasi, pertahankan key yang sama untuk retry. Request tanpa key tetap didukung, tetapi tidak dijamin idempotent.

Ledger menyimpan scope POST:/api/booking:v1, SHA-256 key, SHA-256 payload ternormalisasi dan booking_id FK. Primary key (scope,key_hash) menjamin satu operasi. Payload fingerprint dihitung sebelum nama RS diganti nama canonical Fuseki; perubahan nama client atau field bermakna lain tetap terdeteksi. Whitespace/default JSON dinormalisasi oleh model; bukan fingerprint byte HTTP mentah. Tidak menyimpan nama pasien, telepon, catatan atau response JSON pada ledger. Hash bukan enkripsi: akses DB tetap harus dibatasi.

INSERT ON CONFLICT DO NOTHING RETURNING menunggu transaksi pemilik key. Reservasi ledger, booking, dan pengisian FK terjadi dalam satu transaksi. Request sama membaca booking yang sudah committed; berbeda menghasilkan 409. Collision kode booking rollback seluruh reservasi lalu retry maksimal lima kali. Snapshot BookingResponse dibentuk setelah flush sebelum commit, sehingga tidak ada refresh sesudah commit yang memicu insert ulang. Jika commit acknowledgement hilang, retry keyed mengambil operasi committed. Booking yang dibatalkan tetap replay ke ID yang sama dengan status terkini, bukan membuat booking baru.

Tanggal lampau ditolak hanya untuk operasi baru setelah pengecekan replay, sehingga retry sah hari berikutnya tetap bekerja. Legacy tanpa key masih memiliki risiko duplicate bila client mengulang setelah respons hilang. Frontend lama belum mengirim key; integrasi client harus dilakukan bersama rollout backend agar tidak mengklaim UI sudah terlindungi otomatis.

Scope global endpoint sesuai API publik yang belum memiliki akun. Key harus tidak ditebak dan tidak dibagikan. Mengetahui key saja tidak cukup untuk replay: fingerprint payload harus cocok. Belum ada TTL/cleanup ledger atau idempotency retention policy; jangan menghapus ledger otomatis karena akan membuka duplicate operation. Replay masih melewati validasi input dan lookup Fuseki; outage sumber semantik dapat menghalangi replay sementara, tetapi tidak membuat booking kedua.
